# Robots.txt for TapTest — https://www.taptest.in
#
# taptest.in redirects to www.taptest.in, so the sitemap is named on the www
# host. Pointing at the apex would hand every crawler a redirect to follow.
#
# /login, /register and /profile are deliberately NOT disallowed here. A page
# blocked in robots.txt is never fetched, so Google never sees the noindex on
# it and can still index the bare URL from a link. Those three carry a
# noindex meta tag instead, which removes them for certain.

User-agent: *
Allow: /

# Nothing is served under /api on this host — the API lives on Railway — and
# this keeps it that way.
Disallow: /api/

Sitemap: https://www.taptest.in/sitemap.xml
